<!DOCTYPE html>
<html>
<head>
  <title>TR Football</title>
  <style>
    canvas { background: green; display: block; margin: auto; }
    body { text-align: center; color: white; font-family: Arial; }
    #menu, #difficulty { margin-top: 20px; }
    button { font-size: 20px; margin: 10px; padding: 10px 20px; }
  </style>
</head>
<body>
<h1>TR Football</h1>
<div id="menu">
  <button onclick="selectMode('1v1')">1 vs 1</button>
  <button onclick="selectMode('1vAI')">1 vs AI</button>
</div>
<div id="difficulty" style="display:none;">
  <h2>Zorluk Seç</h2>
  <button onclick="startGame('easy')">Kolay</button>
  <button onclick="startGame('medium')">Orta</button>
  <button onclick="startGame('hard')">Zor</button>
</div>
<canvas id="gameCanvas" width="800" height="600" style="display:none;"></canvas>
<script>
const canvas = document.getElementById("gameCanvas");
const ctx = canvas.getContext("2d");

let mode = null;
let difficulty = null;

// Oyuncular
let player1 = { x:100, y:300, size:20, speed:3 };
let player2 = { x:700, y:300, size:20, speed:3 };
let ai = { x:700, y:300, size:20, speed:2 };

// Top
let ball = { x:400, y:300, size:12, dx:2, dy:1 };

// Skor
let scorePlayer1 = 0, scorePlayer2 = 0;

// Maç süresi
let matchTime = 1800; 
let lastUpdate = Date.now();

// Kaleler
let goalLeft = { x:0, y:200, w:20, h:200 };
let goalRight = { x:780, y:200, w:20, h:200 };

// Klavye kontrol
let keys = {};
document.addEventListener("keydown", e => keys[e.key] = true);
document.addEventListener("keyup", e => keys[e.key] = false);

function selectMode(selectedMode){
  mode = selectedMode;
  document.getElementById("menu").style.display = "none";
  document.getElementById("difficulty").style.display = "block";
}

function startGame(selectedDifficulty){
  difficulty = selectedDifficulty;

  // Zorluk ayarları
  if(difficulty === "easy"){
    ai.speed = 1.5;
    ball.dx = 2; ball.dy = 1;
  } else if(difficulty === "medium"){
    ai.speed = 2.5;
    ball.dx = 3; ball.dy = 2;
  } else if(difficulty === "hard"){
    ai.speed = 4;
    ball.dx = 4; ball.dy = 3;
  }

  document.getElementById("difficulty").style.display = "none";
  canvas.style.display = "block";
  gameLoop();
}

function resetBall(){
  ball.x = canvas.width/2;
  ball.y = canvas.height/2;
  ball.dx = (Math.random()>0.5?ball.dx:-ball.dx);
  ball.dy = (Math.random()>0.5?ball.dy:-ball.dy);
}

function update(){
  // Oyuncu 1 (ok tuşları)
  if(keys["ArrowLeft"]) player1.x -= player1.speed;
  if(keys["ArrowRight"]) player1.x += player1.speed;
  if(keys["ArrowUp"]) player1.y -= player1.speed;
  if(keys["ArrowDown"]) player1.y += player1.speed;

  if(mode === "1v1"){
    // Oyuncu 2 (WASD)
    if(keys["a"]) player2.x -= player2.speed;
    if(keys["d"]) player2.x += player2.speed;
    if(keys["w"]) player2.y -= player2.speed;
    if(keys["s"]) player2.y += player2.speed;
  } else if(mode === "1vAI"){
    // AI topu takip etsin
    if(ai.y < ball.y) ai.y += ai.speed;
    if(ai.y > ball.y) ai.y -= ai.speed;
  }

  // Top hareketi
  ball.x += ball.dx;
  ball.y += ball.dy;

  // Çarpışma (saha sınırları)
  if(ball.x < ball.size) ball.dx *= -1;
  if(ball.x > canvas.width-ball.size) ball.dx *= -1;
  if(ball.y < ball.size || ball.y > canvas.height-ball.size) ball.dy *= -1;

  // Çarpışma (Oyuncu 1)
  let dist1 = Math.hypot(ball.x-player1.x, ball.y-player1.y);
  if(dist1 < player1.size+ball.size){ ball.dx *= -1; }

  if(mode === "1v1"){
    let dist2 = Math.hypot(ball.x-player2.x, ball.y-player2.y);
    if(dist2 < player2.size+ball.size){ ball.dx *= -1; }
  } else if(mode === "1vAI"){
    let distAI = Math.hypot(ball.x-ai.x, ball.y-ai.y);
    if(distAI < ai.size+ball.size){ ball.dx *= -1; }
  }

  // Gol kontrolü
  if(ball.x < goalLeft.x+goalLeft.w && ball.y>goalLeft.y && ball.y<goalLeft.h+goalLeft.y){
    scorePlayer2++;
    resetBall();
  }
  if(ball.x > goalRight.x && ball.y>goalRight.y && ball.y<goalRight.h+goalRight.y){
    scorePlayer1++;
    resetBall();
  }

  // Maç süresi azalt
  let now = Date.now();
  let delta = (now - lastUpdate) / 1000;
  lastUpdate = now;
  matchTime -= delta;
  if(matchTime <= 0){
    alert("Maç Bitti! Skor: Oyuncu1 " + scorePlayer1 + " - Rakip " + scorePlayer2);
    matchTime = 0;
  }
}

function drawBall(){
  ctx.fillStyle="white";
  ctx.beginPath(); ctx.arc(ball.x,ball.y,ball.size,0,Math.PI*2); ctx.fill();
  ctx.strokeStyle="black"; ctx.lineWidth=2; ctx.stroke();
  ctx.fillStyle="black";
  for(let i=0;i<6;i++){
    let angle = i*Math.PI/3;
    let bx = ball.x + Math.cos(angle)*ball.size*0.6;
    let by = ball.y + Math.sin(angle)*ball.size*0.6;
    ctx.beginPath(); ctx.arc(bx,by,3,0,Math.PI*2); ctx.fill();
  }
}

function drawField(){
  ctx.strokeStyle="white";
  ctx.lineWidth=2;
  ctx.strokeRect(0,0,canvas.width,canvas.height);
  ctx.beginPath();
  ctx.moveTo(canvas.width/2,0);
  ctx.lineTo(canvas.width/2,canvas.height);
  ctx.stroke();
  ctx.beginPath();
  ctx.arc(canvas.width/2,canvas.height/2,60,0,Math.PI*2);
  ctx.stroke();
}

function draw(){
  ctx.clearRect(0,0,canvas.width,canvas.height);
  drawField();

  ctx.fillStyle="blue";
  ctx.beginPath(); ctx.arc(player1.x,player1.y,player1.size,0,Math.PI*2); ctx.fill();

  if(mode === "1v1"){
    ctx.fillStyle="red";
    ctx.beginPath(); ctx.arc(player2.x,player2.y,player2.size,0,Math.PI*2); ctx.fill();
  } else if(mode === "1vAI"){
    ctx.fillStyle="red";
    ctx.beginPath(); ctx.arc(ai.x,ai.y,ai.size,0,Math.PI*2); ctx.fill();
  }

  drawBall();

  ctx.fillStyle="yellow";
  ctx.fillRect(goalLeft.x,goalLeft.y,goalLeft.w,goalLeft.h);
  ctx.fillRect(goalRight.x,goalRight.y,goalRight.w,goalRight.h);

  ctx.fillStyle="white";
  ctx.font = "20px Arial";
  ctx.fillText("Oyuncu1: "+scorePlayer1+"  Rakip: "+scorePlayer2, 10,20);

  let minutes = Math.floor(matchTime/60);
  let seconds = Math.floor(matchTime%60);
  ctx.fillText("Süre: "+minutes+":"+String(seconds).padStart(2,"0"), 650,20);
}

function gameLoop(){
  update();
  draw();
  if(matchTime > 0) requestAnimationFrame(gameLoop);
}
</script>
</body>
</html>
